<objective>
Migrate the two central credential-reading choke points — `config.ts`'s
`hasEnvAndCredentials()` (used by every `isPaymentMethodEnabled` call across the whole app)
and `stripe.ts`'s `getStripe()` — onto the resolver built in Plan 08-02, and fix the one
call-site consequence this forces (`checkout.ts` and the Stripe webhook route both consume
the Stripe client, which is becoming async).

Purpose: `hasEnvAndCredentials`/`isPaymentMethodEnabled` is the single gate every payment
method (all 8 gateways) passes through before a customer can even attempt a payment — moving
its credential check onto the resolver is what actually makes ADMN-07's "DB-first with env
fallback" promise true at the point where it matters most.
Output: `config.ts` and `stripe.ts` resolver-backed; `checkout.ts` and the Stripe webhook
route updated for the now-async Stripe client; existing regression suites (config.test.ts,
tests/checkout/*-actions.test.ts) green.
</objective>

    // src/lib/payments/stripe.ts
    // Stripe client singleton — server-only. Phase 8 (D-03a): key sourced from the DB-first/

    import Stripe from "stripe"
    import { getGatewayConfig } from "./gateway-credentials"

    let _stripe: Stripe | undefined

    export async function getStripe(): Promise<Stripe> {
      if (!_stripe) {
        const cfg = await getGatewayConfig("stripe")
        const key = cfg.secret_key
        if (!key) throw new Error("Stripe secret_key is not configured (DB or env)")
        _stripe = new Stripe(key, { typescript: true })
      }
      return _stripe
    }
