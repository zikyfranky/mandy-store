    type RawRow = { provider: string; field: string; value: string; isSecret: boolean }

    // Request-scoped batched read — ONE query regardless of how many fields/providers are
    // checked this request (mirrors getStoreSettingsMap exactly). React.cache is request-scoped
    // only — no revalidateTag/revalidatePath needed after writes (Pitfall 6).
    export const getGatewayCredentialsMap = cache(async (): Promise<Record<string, Record<string, RawRow>>> => {
      const rows = await db.select().from(gatewayCredentials)
      const map: Record<string, Record<string, RawRow>> = {}
      for (const r of rows) {
        map[r.provider] ??= {}
        map[r.provider][r.field] = r
      }
      return map
    })

    /**
     * Resolves ALL fields for a provider. DB value wins if present and NON-EMPTY; else falls
     * back to process.env[envVar]. A field that resolves to "" or undefined from BOTH sources
     * is OMITTED from the result — callers must treat a missing key as "not configured"
     * (fail-closed, D-03). decryptSecret's throw (bad/rotated key, tampered ciphertext)
     * propagates OUT of this function uncaught (D-07) — do not wrap in try/catch here.
     */
    export async function getGatewayConfig(provider: string): Promise<Record<string, string>> {
      const manifest = FIELD_MANIFEST[provider] ?? []
      const map = await getGatewayCredentialsMap()
      const providerRows = map[provider] ?? {}

      const result: Record<string, string> = {}
      for (const spec of manifest) {
        const dbRow = providerRows[spec.field]
        let value: string | undefined
        if (dbRow && dbRow.value !== "") {
          value = dbRow.isSecret ? decryptSecret(dbRow.value) : dbRow.value
        } else {
          value = process.env[spec.envVar] || undefined
        }
        if (value) result[spec.field] = value
      }
      return result
    }

    /** Convenience single-field accessor for call sites that only need one value. */
    export async function getGatewayCredential(provider: string, field: string): Promise<string | undefined> {
      const config = await getGatewayConfig(provider)
      return config[field]
    }

## Trust Boundaries

| Boundary | Description |
|----------|--------------|
| DB row → resolver → caller | A stored ciphertext crosses back into plaintext only inside `getGatewayConfig`; callers receive plaintext values already decrypted, never the envelope |
| `getGatewayCredentialsMap` → any consumer | Request-scoped cache only — no cross-request staleness protection assumed or needed |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-05 | Information Disclosure | `getGatewayConfig` | mitigate | `decryptSecret`'s throw propagates uncaught — never coerced into `undefined`/"not configured" (Pitfall 4); test suite asserts the rejection surfaces |
| T-08-06 | Tampering / Denial of Service | Empty-string DB row handling | mitigate | `dbRow.value !== ""` check before treating a row as a DB hit (Pitfall 3) — an admin clearing a field without deleting the row can never silently mask the env fallback |
| T-08-07 | Elevation of Privilege | Fail-closed field omission | mitigate | A field resolving empty from both DB and env is omitted from the result object entirely (never `""`) — every caller in later plans checks truthiness, never treats an empty string as configured |
</threat_model>
