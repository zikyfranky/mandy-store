import "server-only"
// PAY-04 / D-13: server-only payment-method toggles. Enforced in actions/route handlers,
// never just the UI. PayPal deferred (D-05a) — defaults OFF.

export type PaymentMethod = "stripe" | "paypal" | "bank_transfer"

export function isPaymentMethodEnabled(method: PaymentMethod): boolean {
  switch (method) {
    case "stripe":
      return process.env.ENABLE_STRIPE === "true" && !!process.env.STRIPE_SECRET_KEY
    case "paypal":
      return process.env.ENABLE_PAYPAL === "true"
        && !!process.env.PAYPAL_CLIENT_ID && !!process.env.PAYPAL_CLIENT_SECRET
    case "bank_transfer":
      return process.env.ENABLE_BANK_TRANSFER === "true"
        && !!process.env.BANK_NAME && !!process.env.BANK_ACCOUNT_NUMBER && !!process.env.BANK_ACCOUNT_NAME
  }
}

/** List of enabled methods for the UI (server-computed; passed as a prop). */
export function enabledPaymentMethods(): PaymentMethod[] {
  return (["stripe", "paypal", "bank_transfer"] as const).filter(isPaymentMethodEnabled)
}
```

import { orders } from "@/lib/db/schema"
import { eq } from "drizzle-orm"

export async function getOrderByToken(token: string) {
  if (!token) return undefined
  return db.query.orders.findFirst({
    where: eq(orders.guestToken, token),
    with: { items: true },
  })
}

export async function getOrderByStripePaymentIntentId(piId: string) {
  return db.query.orders.findFirst({
    where: eq(orders.stripePaymentIntentId, piId),
    with: { items: true },
  })
}

export async function getOrderStatusByToken(token: string): Promise<string | null> {
  const order = await db.query.orders.findFirst({
    where: eq(orders.guestToken, token),
    columns: { status: true },
  })
  return order?.status ?? null
}

import "server-only"
import { Resend } from "resend"
import { OrderConfirmationEmail } from "@/emails/OrderConfirmationEmail"

const resend = new Resend(process.env.RESEND_API_KEY!)

export interface OrderConfirmationInput {
  customerEmail: string; reference: string; guestToken: string
  items: Array<{ productName: string; variantLabel: string; quantity: number; lineTotalKobo: number }>
  subtotalKobo: number; shippingKobo: number; totalKobo: number
}

export async function sendOrderConfirmationEmail(order: OrderConfirmationInput): Promise<void> {
  const { error } = await resend.emails.send({
    from: "Mandy Store <onboarding@resend.dev>",
    to: order.customerEmail,
    subject: `Order confirmed — ${order.reference}`,
    react: OrderConfirmationEmail({
      orderRef: order.reference, orderToken: order.guestToken, customerEmail: order.customerEmail,
      items: order.items, subtotalKobo: order.subtotalKobo, shippingKobo: order.shippingKobo, totalKobo: order.totalKobo,
    }),
  })
  if (error) throw new Error(`Failed to send order confirmation email: ${error.message}`)
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| env config → enabled methods | Server config decides which methods exist; client must never override |
| order token → order data | Token-bearing lookup must be unguessable |
| bank secrets → email | Bank details flow only through server-only sender to the customer's inbox |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-05 | Elevation of Privilege | isPaymentMethodEnabled | mitigate | toggle is server-only (`import "server-only"`); read in actions/route handlers (Plans 03/04), never trusted from UI; PayPal forced OFF (no client_id) |
| T-04-06 | Information Disclosure | guest token entropy | mitigate | generateGuestToken uses crypto.getRandomValues(32 bytes)=256-bit; never Math.random; orders.guestToken is unique-indexed (Plan 01) |
| T-04-07 | Information Disclosure | bank account env vars | mitigate | send-bank-transfer-instructions.ts is `server-only`; reads BANK_* (non-NEXT_PUBLIC_); details only emailed, never returned in any client payload |
| T-04-08 | Spoofing | Stripe secret key | mitigate | stripe.ts is `server-only`; STRIPE_SECRET_KEY never bundled to client; only the publishable key is NEXT_PUBLIC_ |
| T-04-09 | Tampering | order reference collision | mitigate | generateUniqueOrderRef retries against unique DB constraint (10 attempts) before throwing |
</threat_model>
