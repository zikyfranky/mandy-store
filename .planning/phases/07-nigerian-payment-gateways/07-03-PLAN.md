export interface GatewayAdapter {
  readonly provider: "paystack" | "flutterwave"
  initPayment(params: { email; amountKobo; reference; orderId; customerName; phone? }): Promise<Record<string, unknown>>

}
export interface VerifyResult { status: "confirmed"|"failed"|"pending"; amountKobo: number; currency: string; reference: string }

Flutterwave API (RESEARCH Patterns 2, 5; Code Examples; Pitfalls 1,5,7,8):
- No server-side "initialize" — params assembled server-side and passed to the client FlutterwaveCheckout(). amount is NAIRA: `amount: amountKobo / 100`. Include public_key = NEXT_PUBLIC_FLUTTERWAVE_PUBLIC_KEY, tx_ref, currency:"NGN", customer:{email,name,phone_number}, meta:{orderId}, customizations.
- Verify: GET https://api.flutterwave.com/v3/transactions/:id/verify, Bearer FLUTTERWAVE_SECRET_KEY → data.status ("successful", NOT "success"), data.amount (NAIRA), data.currency, data.tx_ref.
- Refund: POST https://api.flutterwave.com/v3/transactions/:id/refund, body { amount?: naira } (min ₦100; omit for full).
- Webhook: header `flutterwave-signature` (HMAC SHA-256 base64 of raw body using FLUTTERWAVE_WEBHOOK_HASH) with `verif-hash` fallback. Event type "charge.completed", data.status "successful", data.tx_ref, data.id (numeric txn id), data.amount (NAIRA). eventId = "flutterwave:"+data.tx_ref.
